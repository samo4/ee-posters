  \begin{eqgroup}[Black: gain $A$, feedback $\beta$, loop gain $L=A\beta$]
  \[ A_f=\frac{A}{1+L}\ \xrightarrow{\ L\,\gg\,1\ }\ \frac{1}{\beta} \]
  \[ \frac{dA_f}{A_f}=\frac{1}{1+L}\,\frac{dA}{A} \]
  \end{eqgroup}
  \begin{eqgroup}[Stability: look at crossover, $|L(j\omega_c)|=1$]
  \[ \mathrm{PM}=180^\circ+\angle L(j\omega_c) \]
  \end{eqgroup}}%
