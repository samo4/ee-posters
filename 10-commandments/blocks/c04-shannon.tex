\begin{commandment}[slot]{IV}{Shannon}{Thou shalt not sample below twice the bandwidth.}{ADCs, anti-aliasing filters, compression, error-correcting codes, adaptive QAM in Wi-Fi and 5G.}
\body{%
  \begin{eqgroup}[Sampling \& quantisation]
  \[ f_s>2B \qquad \sigma_q^2=\frac{\Delta^2}{12}\ \ \text{\small\color{muted}(6 dB per bit)} \]
  \end{eqgroup}
  \begin{eqgroup}[Capacity \& information]
  \[ C=B\log_2\!\Big(1+\frac{S}{N}\Big)\ \ \text{bit/s} \]
  \[ H=-\sum_i p_i\log_2 p_i \]
  \end{eqgroup}}%
